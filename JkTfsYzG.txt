local function getCrateRarity(crateName)
local lowerName = crateName:lower()
for rarity, _ in pairs(RARITY_ORDER) do
if lowerName:find(rarity:lower()) then
return rarity
end
end
return "Common"
end

-- Halo Detection Function (Detects 3D halo models, visual effects, highlights, or attachments)
local function hasHaloEffect(crate)
if crate:FindFirstChild("Halo", true) or crate:FindFirstChild("halo", true) then
return true
end

for _, child in ipairs(crate:GetDescendants()) do
if child:IsA("ParticleEmitter") or child:IsA("Beam") or child:IsA("Highlight") or child:IsA("PointLight") then
local name = child.Name:lower()
if name:find("halo") or name:find("blessing") or name:find("yellow") then
return true
end
end
if child:IsA("BasePart") or child:IsA("MeshPart") then
if child.Name:lower():find("halo") then
return true
end
end
end
return false
end

-- Helper function to make frames draggable
local function makeDraggable(gui)
local dragging, dragInput, dragStart, startPos

local function update(input)
local delta = input.Position - dragStart
gui.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
end

gui.InputBegan:Connect(function(input)
if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
dragging = true
dragStart = input.Position
startPos = gui.Position

input.Changed:Connect(function()
if input.UserInputState == Enum.UserInputState.End then
dragging = false
end
end)
end
end)

gui.InputChanged:Connect(function(input)
if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
dragInput = input
end
end)

UserInputService.InputChanged:Connect(function(input)
if input == dragInput and dragging then
update(input)
end
end)
end

-- Steal Action
local function stealCrate(crate, btn)
local hrp = character and character:FindFirstChild("HumanoidRootPart")
if not hrp then return end

if not safeZoneCFrame then
safeZoneBtn.Text = "⚠️ Set Safe Zone First!"
safeZoneBtn.BackgroundColor3 = Color3.fromRGB(200, 50, 50)
task.wait(1)
safeZoneBtn.Text = "📍 Set Safe Zone: [ Not Set ]"
safeZoneBtn.BackgroundColor3 = Color3.fromRGB(0, 120, 215)
return
end

if not crate or not crate.Parent then return end

local targetPart = crate:IsA("Model") and (crate.PrimaryPart or crate:FindFirstChildWhichIsA("BasePart")) or (crate:IsA("BasePart") and crate)
if not targetPart then return end

btn.Text = "Stealing..."
btn.BackgroundColor3 = Color3.fromRGB(200, 140, 0)

-- 1. Teleport to crate position
hrp.CFrame = targetPart.CFrame * CFrame.new(0, 0, 1.5)

task.wait(1.2)

-- 2. Instant Prompt Trigger
for _, prompt in ipairs(crate:GetDescendants()) do
if prompt:IsA("ProximityPrompt") then
prompt.HoldDuration = 0
prompt.MaxActivationDistance = 9999
pcall(function()
fireproximityprompt(prompt)
end)
end
end

task.wait(0.8)

-- 3. Teleport back to Safe Zone
hrp.CFrame = safeZoneCFrame

btn.Text = "Done!"
btn.BackgroundColor3 = Color3.fromRGB(35, 160, 80)
task.wait(1)
if btn and btn.Parent then
btn.Text = "TP & Steal"
btn.BackgroundColor3 = Color3.fromRGB(45, 45, 60)
end
end

if targetPart then
local rarity = getCrateRarity(crate.Name)
local isHaloCrate = hasHaloEffect(crate) or rarity == "Blessinghalo"
local rank = isHaloCrate and 0 or (RARITY_ORDER[rarity] or 99)

rarity = "Blessinghalo"
end
